// 核心模板：读入输出 + 二分答案三板斧 + 01 背包
#include <bits/stdc++.h>
using namespace std;
int n, m;

// 1. 文件读写（必须）
int main() {
    freopen("problem.in", "r", stdin);
    freopen("problem.out", "w", stdout);

    // 2. 二分答案：确定 check(mid) 含义 → 证明单调性 → 处理边界
    int l = 0, r = 1e9, ans = 0;
    while (l <= r) {
        int mid = (l + r) >> 1;
        if (check(mid)) ans = mid, l = mid + 1;
        else r = mid - 1;
    }

    // 3. 01 背包：每个数及其和不大时识别为背包
    int w[105], v[105], dp[10005];
    for (int i = 1; i <= n; ++i)
        for (int j = m; j >= w[i]; --j)
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
    return 0;
}
